local Config = {
    Speed = 70,
    Minimum = 16,
    Maximum = 200,
    Enabled = false,
    Particles = true,
}

local FILE_NAME = "WhitezMonoSpeed.json"
local SaveVersion = 0

local function ClampSpeed(value)
    value = tonumber(value)

    if not value or value ~= value then
        return Config.Speed
    end

    return math.clamp(
        math.floor(value + 0.5),
        Config.Minimum,
        Config.Maximum
    )
end

pcall(function()
    if type(readfile) ~= "function" then
        return
    end

    if type(isfile) == "function" and not isfile(FILE_NAME) then
        return
    end

    local saved = HttpService:JSONDecode(readfile(FILE_NAME))

    if type(saved) == "table" then
        Config.Speed = ClampSpeed(saved.Speed)

        if type(saved.Particles) == "boolean" then
            Config.Particles = saved.Particles
        end
    end
end)

local function SaveNow()
    if type(writefile) ~= "function" then
        return
    end

    pcall(function()
        writefile(FILE_NAME, HttpService:JSONEncode({
            Speed = Config.Speed,
            Particles = Config.Particles,
        }))
    end)
end

local function ScheduleSave()
    SaveVersion += 1
    local version = SaveVersion

    task.delay(0.5, function()
        if Alive and version == SaveVersion then
            SaveNow()
        end
    end)
end

local function New(className, properties, parent)
    local object = Instance.new(className)

    for property, value in pairs(properties or {}) do
        object[property] = value
    end

    object.Parent = parent
    return object
end

local function Corner(parent, radius)
    return New("UICorner", {
        CornerRadius = UDim.new(0, radius or 12),
    }, parent)
end

local function Stroke(parent, transparency, thickness)
    return New("UIStroke", {
        Color = Color3.fromRGB(255, 255, 255),
        Transparency = transparency or 0.84,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function Gradient(parent, top, bottom)
    return New("UIGradient", {
        Rotation = 90,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, top),
            ColorSequenceKeypoint.new(1, bottom),
        }),
    }, parent)
end

local function Tween(object, properties, duration)
    if not Alive or not object.Parent then
        return
    end

    local previous = ActiveTweens[object]
    if previous then
        pcall(function()
            previous:Cancel()
        end)
    end

    local animation = TweenService:Create(
        object,
        TweenInfo.new(
            duration or 0.18,
            Enum.EasingStyle.Quad,
            Enum.EasingDirection.Out
        ),
        properties
    )

    ActiveTweens[object] = animation
    animation:Play()

    task.delay((duration or 0.18) + 0.1, function()
        if ActiveTweens[object] == animation then
            ActiveTweens[object] = nil
        end
    end)

    return animation
end

local function ClampPosition(object, x, y)
    local viewport = Viewport()
    local size = object.AbsoluteSize

    return math.clamp(x, 6, math.max(6, viewport.X - size.X - 6)),
        math.clamp(y, 6, math.max(6, viewport.Y - size.Y - 6))
end

local function KeepOnScreen(object)
    local x, y = ClampPosition(
        object,
        object.Position.X.Offset,
        object.Position.Y.Offset
    )

    object.Position = UDim2.fromOffset(x, y)
end

local InitialLayout = true

local function UpdateLayout()
    local viewport = Viewport()
    local scale = math.clamp(
        math.min((viewport.X - 24) / 360, (viewport.Y - 24) / 348),
        0.35,
        1
    )

    WindowScale.Scale = scale
    FloatingScale.Scale = math.clamp(scale, 0.8, 1)
    ToastScale.Scale = math.min(1, (viewport.X - 20) / 290)

    if InitialLayout then
        InitialLayout = false

        Window.Position = UDim2.fromOffset(
            math.max(6, (viewport.X - 360 * scale) / 2),
            math.max(6, (viewport.Y - 348 * scale) / 2)
        )

        Floating.Position = UDim2.fromOffset(
            15,
            math.max(8, viewport.Y * 0.5 - 24)
        )
    else
        KeepOnScreen(Window)
        KeepOnScreen(Floating)
    end
end

local DragStates = {}

local function MakeDraggable(handle, target)
    local state = {
        Active = false,
        Moved = false,
        Input = nil,
        Start = Vector2.zero,
        Origin = Vector2.zero,
    }

    table.insert(DragStates, state)

    Connect(handle.InputBegan, function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then
            return
        end

        if state.Active then
            return
        end

        state.Active = true
        state.Moved = false
        state.Input = input
        state.Start = Vector2.new(input.Position.X, input.Position.Y)
        state.Origin = Vector2.new(
            target.Position.X.Offset,
            target.Position.Y.Offset
        )
    end)

    Connect(UserInputService.InputChanged, function(input)
        if not state.Active then
            return
        end

        local touch = state.Input.UserInputType == Enum.UserInputType.Touch

        if touch and input ~= state.Input then
            return
        end

        if not touch
            and input.UserInputType ~= Enum.UserInputType.MouseMovement then
            return
        end

        local current = Vector2.new(input.Position.X, input.Position.Y)
        local delta = current - state.Start

        if delta.Magnitude > 6 then
            state.Moved = true
        end

        if state.Moved then
            local x, y = ClampPosition(
                target,
                state.Origin.X + delta.X,
                state.Origin.Y + delta.Y
            )

            target.Position = UDim2.fromOffset(x, y)
        end
    end)

    Connect(UserInputService.InputEnded, function(input)
        if input == state.Input
            or (
                state.Input
                and state.Input.UserInputType == Enum.UserInputType.MouseButton1
                and input.UserInputType == Enum.UserInputType.MouseButton1
            ) then
            state.Active = false
            state.Input = nil
        end
    end)

    return state
end

local Humanoid
local OriginalSpeed
local BoundCharacter
local CharacterVersion = 0

local function RestoreSpeed()
    if Humanoid and Humanoid.Parent and OriginalSpeed then
        pcall(function()
            Humanoid.WalkSpeed = OriginalSpeed
        end)
    end

    OriginalSpeed = nil
end

local function ApplySpeed()
    if not Config.Enabled
        or not Humanoid
        or not Humanoid.Parent
        or Humanoid.Health <= 0 then
        return
    end

    pcall(function()
        if OriginalSpeed == nil then
            OriginalSpeed = Humanoid.WalkSpeed
        end

        if math.abs(Humanoid.WalkSpeed - Config.Speed) > 0.01 then
            Humanoid.WalkSpeed = Config.Speed
        end
    end)
end

local function RefreshStatus()
    if not Config.Enabled then
        Status.Text = "DESATIVADO"
        StatusDot.BackgroundColor3 = Color3.fromRGB(110, 110, 118)
    elseif not Humanoid or not Humanoid.Parent or Humanoid.Health <= 0 then
        Status.Text = "ATIVO · AGUARDANDO PERSONAGEM"
        StatusDot.BackgroundColor3 = Color3.fromRGB(180, 180, 185)
    else
        Status.Text = "ATIVO · " .. Config.Speed .. " STUDS/S"
        StatusDot.BackgroundColor3 = Color3.new(1, 1, 1)
    end
end

local function BindCharacter(character)
    if BoundCharacter == character then
        return
    end

    CharacterVersion += 1
    local version = CharacterVersion

    RestoreSpeed()
    Humanoid = nil
    BoundCharacter = character
    RefreshStatus()

    task.spawn(function()
        local found = character:FindFirstChildOfClass("Humanoid")
            or character:WaitForChild("Humanoid", 15)

        if not Alive
            or version ~= CharacterVersion
            or Player.Character ~= character
            or not found
            or not found:IsA("Humanoid") then
            return
        end

        Humanoid = found
        OriginalSpeed = nil
        ApplySpeed()
        RefreshStatus()
    end)
end

local function SetEnabled(enabled)
    Config.Enabled = enabled

    if enabled then
        ApplySpeed()
    else
        RestoreSpeed()
    end

    PaintSpeedToggle(enabled)
    RefreshStatus()

    Notify(enabled and "WalkSpeed ativado." or "Velocidade original restaurada.")
end

local function RefreshSlider()
    local alpha = (Config.Speed - Config.Minimum)
        / (Config.Maximum - Config.Minimum)

    SliderFill.Size = UDim2.fromScale(alpha, 1)
    SliderKnob.Position = UDim2.fromScale(alpha, 0.5)

    if not SpeedInput:IsFocused() then
        SpeedInput.Text = tostring(Config.Speed)
    end
end

local function SetSpeed(value)
    Config.Speed = ClampSpeed(value)
    RefreshSlider()
    ApplySpeed()
    RefreshStatus()
    ScheduleSave()
end

local function SetSliderFromX(x)
    local width = SliderTrack.AbsoluteSize.X

    if width <= 0 then
        return
    end

    local alpha = math.clamp(
        (x - SliderTrack.AbsolutePosition.X) / width,
        0,
        1
    )

    SetSpeed(Config.Minimum + alpha * (Config.Maximum - Config.Minimum))
end

local Randomizer = Random.new()
local Particles = {}

for index = 1, 21 do
    local size = Randomizer:NextInteger(1, 3)
    local particle = New("Frame", {
        Name = "Particle" .. index,
        Size = UDim2.fromOffset(size, size),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BackgroundTransparency = Randomizer:NextNumber(0.45, 0.82),
        BorderSizePixel = 0,
        ZIndex = 3,
    }, ParticleLayer)

    Corner(particle, 5)

    local data = {
        Object = particle,
        X = Randomizer:NextNumber(0.03, 0.97),
        Y = Randomizer:NextNumber(0.02, 0.98),
        Speed = Randomizer:NextNumber(0.022, 0.058),
        Phase = Randomizer:NextNumber(0, math.pi * 2),
        Opacity = Randomizer:NextNumber(0.5, 0.78),
    }

    particle.Position = UDim2.fromScale(data.X, data.Y)
    table.insert(Particles, data)
end

local SpeedTimer = 0
local ParticleTimer = 0
local LayoutTimer = 0
local Elapsed = 0
local LastViewport = Viewport()

Connect(RunService.Heartbeat, function(deltaTime)
    if not Alive then
        return
    end

    Elapsed += deltaTime
    SpeedTimer += deltaTime
    ParticleTimer += deltaTime
    LayoutTimer += deltaTime

    -- Atualização moderada: não força WalkSpeed a cada frame.
    if SpeedTimer >= 0.12 then
        SpeedTimer = 0
        ApplySpeed()
        RefreshStatus()
    end

    if LayoutTimer >= 0.4 then
        LayoutTimer = 0

        local viewport = Viewport()
        if viewport ~= LastViewport then
            LastViewport = viewport
            UpdateLayout()
        end
    end

    -- Partículas pausam enquanto o painel está minimizado.
    if ParticleTimer >= 1 / 30 then
        local step = math.min(ParticleTimer, 0.1)
        ParticleTimer = 0

        if Window.Visible and Config.Particles then
            for _, data in ipairs(Particles) do
                data.Y -= data.Speed * step

                if data.Y < -0.02 then
                    data.Y = 1.02
                    data.X = Randomizer:NextNumber(0.03, 0.97)
                end

                local sway = math.sin(Elapsed * 0.65 + data.Phase) * 0.012

                data.Object.Position = UDim2.fromScale(
                    data.X + sway,
                    data.Y
                )

                data.Object.BackgroundTransparency = math.clamp(
                    data.Opacity + math.sin(Elapsed + data.Phase) * 0.09,
                    0.3,
                    0.9
                )
            end
        end
    end
end)

Cleanup = function()
    if not Alive then
        return
    end

    Alive = false
    CharacterVersion += 1
    SaveVersion += 1
    ToastVersion += 1

    SaveNow()
    Config.Enabled = false
    RestoreSpeed()

    for _, connection in ipairs(Connections) do
        pcall(function()
            connection:Disconnect()
        end)
    end
    table.clear(Connections)

    for _, animation in pairs(ActiveTweens) do
        pcall(function()
            animation:Cancel()
        end)
    end
    table.clear(ActiveTweens)
    table.clear(Particles)
    table.clear(DragStates)

    pcall(function()
        Gui:Destroy()
    end)

    if Env[INSTANCE_KEY] == App then
        Env[INSTANCE_KEY] = nil
    end
end

App.Destroy = Cleanup

Connect(Close.Activated, Cleanup)

Connect(Gui.AncestryChanged, function()
    if Alive and Gui.Parent == nil then
        Cleanup()
    end
end)
